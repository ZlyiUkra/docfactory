# Лише те, що потрібно.

# Python 3.10+. Склад той самий, що в vite і nextjs, плюс три пакети читача cncf-curriculum
# (програми іспитів CNCF, що видаються лише як PDF) — внизу. Решта читачів репозиторію й
# релізів (markdown, HTML, YAML, JSON) — на стандартній бібліотеці. Версії й верхні межі
# — ті самі, що в instances/react/requirements.txt.

anthropic>=0.40,<1
python-dotenv>=1.0.0

# MCP-сервер і клієнт. У SDK 2.0 клас FastMCP перейменували на MCPServer —
# код підтримує обидві версії.
mcp[cli]>=2.1,<3

# Пошук за змістом (необовʼязковий). ONNX замість torch: імпортується за секунди,
# без тисяч дрібних файлів. Без нього працює пошук лише по словах.
fastembed>=0.8,<1

# Читач cncf-curriculum. Перевірено 02.10.2026 на Python 3.14.6 з pypdf 5.9.0 (03.10 — 6.19.0),
# pypdfium2 5.13.0 і rapidocr-onnxruntime 1.2.3.
#
# Текст PDF з позиціями фрагментів (дві колонки, старі файли «по літері»). Чистий
# Python, ~2 МБ. 6.x — бо 5.x більше не латають: у 5.9 лишалось 49 відомих вад розбору
# (зациклення й вичерпання пам'яті на пошкоджених PDF). 6.x інакше віддає гліфи файлів,
# набраних по літері, і читач під це підлаштовано: 03.10.2026 на тих самих 50 PDF текст
# 45 програм збігся з 5.9 до символу, 5 вийшли кращими (рядки й пункти на своїх місцях).
pypdf>=6,<7
# Чотири програми (CAPA, CCA, CGOA, стара ICA) — картинки без текстового шару, їх
# сторінки розпізнаються. pypdfium2 малює сторінку (~6 МБ, своє колесо PDFium);
# rapidocr-onnxruntime читає намальоване на onnxruntime, що вже стоїть для fastembed
# (~14 МБ з моделями), і тягне opencv-python (~117 МБ — найважче тут), shapely,
# pyclipper, six — разом ~145 МБ. Без системного Tesseract, лише pip. Новіші за 1.2
# rapidocr-onnxruntime на Python 3.13+ не ставляться, тож межа — на перевіреній лінії.
pypdfium2>=4,<6
rapidocr-onnxruntime>=1.2,<1.3
