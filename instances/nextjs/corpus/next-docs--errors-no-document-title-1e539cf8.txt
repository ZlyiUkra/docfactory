# `<title>` should not be used in _document.js's `<Head>`
# джерело: https://github.com/vercel/next.js/blob/7.0.3/errors/no-document-title.md
# отримано: 2026-10-01
# версія: 7.0.3, 7.0.3-alpha.5, 7.0.3-alpha.4, 7.0.3-alpha.3, 7.0.3-alpha.2, 7.0.3-alpha.1, 7.0.3-alpha.0, 7.0.2, 7.0.2-canary, 7.0.2-alpha.3, 7.0.2-alpha.2, 7.0.2-alpha.1, 7.0.2-alpha.0, 7.0.1, 7.0.1-canary

#### Why This Error Occurred

Adding `<title>` in `pages/_document.js` will lead to unexpected results with `next/head` since `_document.js` is only rendered on the initial pre-render.

#### Possible Ways to Fix It

Set `<title>` in `pages/_app.js` instead :

```js
// pages/_app.js
import App, {Container} from 'next/app'
import Head from 'next/head'
import React from 'react'

export default class MyApp extends App {
  static async getInitialProps ({ Component, router, ctx }) {
    let pageProps = {}

    if (Component.getInitialProps) {
      pageProps = await Component.getInitialProps(ctx)
    }

    return {pageProps}
  }

  render () {
    const {Component, pageProps} = this.props
    return <Container>
      <Head>
        <title>My new cool app</title>
      </Head>
      <Component {...pageProps} />
    </Container>
  }
}
```
