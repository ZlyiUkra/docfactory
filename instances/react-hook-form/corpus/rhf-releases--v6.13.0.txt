🌟 custom hook: useController (#3488)

```tsx
import React from "react";
import { useController } from "react-hook-form";

function Input({ control, name }) {
  const {
    field,
    meta: { invalid, isTouched, isDirty },
  } = useController({
    name,
    control,
    rules: { required: true },
  });

  return <input {...field} />;
}
```
