Input

`name="firstName"` | `{ firstName: 'value'}`

`name="name.firstName"` | `{ name: { firstName: 'value' } }`

`name="name.firstName[0]"` | `{ name: { firstName: [ 'value' ] } }`

`ref` `React.RefObject` | React element ref

`required` ``` string | { value: boolean, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ ${ isStandard ? "required: true" : `required: 'error message' // <p>error message</p>` } }) } /> ```

`maxLength` ``` { value: number, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ ${ isStandard ? "maxLength: 2" : `maxLength : { value: 2, message: 'error message' // <p>error message</p> }` } }) } /> ```

`minLength` ``` { value: number, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ minLength: ${ isStandard ? 1 : `{ value: 1, message: 'error message' // <p>error message</p> }` } }) } /> ```

`max` ``` { value: number, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ max: ${ isStandard ? 3 : `{ value: 3, message: 'error message' // <p>error message</p> }` } }) } /> ```

`min` ``` { value: number, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ min: ${ isStandard ? 3 : `{ value: 3, message: 'error message' // <p>error message</p> }` } }) } /> ```

`pattern` ``` { value: RegExp, message: string | React.ReactElement } ``` | ``` <input name="test" ref={ register({ pattern: ${ isStandard ? "/[A-Za-z]{3}/" : `{ value: /[A-Za-z]{3}/, message: 'error message' // <p>error message</p> }` } }) } /> ```

`validate` `Function | Object` | ``` <input name="test" ref={ register({ validate: ${ isStandard ? `value => value === '1'` : `value => value === '1' || 'error message' // <p>error message</p>` } }) } /> // object of callback functions <input name="test1" ref={ register({ validate: ${ isStandard ? `{ positive: value => parseInt(value, 10) > 0, lessThanTen: value => parseInt(value, 10) < 10, asyncValidate: async value => await fetch(url) }` : `{ positive: value => parseInt(value, 10) > 0 || 'should be greater than 0', lessThanTen: value => parseInt(value, 10) < 10 || 'should be lower than 10', // you can do asynchronous validation as well asyncValidate: async value => await fetch(url) || 'error message' // <p>error message</p> }` } }) } /> ```

valueAsNumber: `boolean` | Returns a Number normally. If something goes wrong `NaN` will be returned. | ``` <input name="test" type="number" ref={ register({ valueAsNumber: true, }) } /> ```

valueAsDate: `boolean` | Returns a Date normally. If something goes wrong `null` will be returned. | ``` <input name="test" type="date" ref={ register({ valueAsDate: true, }) } /> ```

setValueAs: `<T>(value: any) => T` | Return input value by running through the function. | ``` <input name="test" ref={ register({ setValueAs: (value) => parseInt(value), }) } /> ```
