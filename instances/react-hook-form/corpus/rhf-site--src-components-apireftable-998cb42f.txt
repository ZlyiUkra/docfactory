`required` | ``` <input name="test" ref={ register({ ${isStandard ? "required: true" : `required: 'error message'`} }) } /> ```

`maxLength` ``` { value: number, message: string } ``` | ``` <input name="test" ref={ register({ ${ isStandard ? "maxLength: 2" : `maxLength : { value: 2, message: 'error message' }` } }) } /> ```

`min` ``` { value: number, message: string } ``` | ``` <input name="test" ref={ register({ min: ${ isStandard ? 3 : `{ value: 3, message: 'error message' }` } }) } /> ```

`validate` `Function | Object` | ``` // callback function <input name="test" ref={ register({ validate: ${ isStandard ? `value => value === '1'` : `value => value === '1' || 'error message';` } }) } /> // object of callback functions <input name="test1" ref={ register({ validate: ${ isStandard ? `{ positive: value => parseInt(value) > 0, lessThanTen: value => parseInt(value) < 10 }` : `{ positive: value => parseInt(value) > 0 || 'should be greater than 0', lessThanTen: value => parseInt(value) < 100 || 'should be greater than 10' }` } }) } /> // you can do asynchronous validation as well <input name="test2" ref={ register({ validate: ${ isStandard ? `async value => await fetch(url)` : `async value => await fetch(url) || 'error message';` } }) } /> ```
