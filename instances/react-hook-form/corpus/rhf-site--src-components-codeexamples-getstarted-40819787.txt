import React from "react";

  male = "male",

  "First Name": string;
  Age: number;
}

  label: Path<IFormValues>;
  register: UseFormRegister<IFormValues>;
  required: boolean;
};

    <label>{label}</label>
    <input {...register(label, { required })} />
  </>

  </>

      <Input label="First Name" register={register} required />

    setValue("AntdInput", e.target.value);
