import React from 'react'
import useForm from 'react-hook-form

function YourForm() {
  const { register, handleSubmit, watch, errors } = useForm()
  const onSubmit = data => {
    console.log(data)
  }; // your form submit function which will invoke after successful validation

  console.log(watch('example')) // you can watch individual input by pass the name of the input

  return (
    {/* handleSubmit will validation your inputs before calling onSubmit */}
    <form onsubmit={handleSubmit(onSubmit)}>
      {/* you will have to register your input into react-hook-form, by invoke the register function with ref as the argument */}
      <input
        type="text"
        name="example"
        ref={ref => {
          register({ ref });
        }}
      />
      {/* include validation with required field or other standard html validation rules */}
      <input
        type="text"
        name="exampleRequired"
        ref={ref => {
          register({ ref, required: true, max: 10 });
        }}
      />
      {/* errors will return true if particular field validation is invalid  */}
      {errors.example && '<span>This field is required</span>'}
      <input type="submit" />
    </form>
  );
}

import React from 'react'
import useForm from 'react-hook-form

function YourForm() {
  const { register, errors, handleSubmit } = useForm();
  const onSubmit = (data) => {};

  return (
    <form onsubmit={handleSubmit(onSubmit)}>
      <input type="text" name="textInput" ref={ ref => { register({ ref, required, maxLength: 50 })} } />
      {errors.textInput && errors.textInput.required && 'Your input is required'}
      {errors.textInput && errors.textInput.maxLength && 'Your input exceed maxLength'}

      <input type="number" name="numberInput" ref={ ref => { register({ ref, min: 50 })} } />
      {errors.numberInput && 'Your input required to be more than 50'}
      <input type="submit" />
    </form>
  )
}

## API

## useForm: ({ mode: 'onSubmit' | 'onBlur' | 'onChange' }):{ register, errors, handleSubmit, watch }

You need to initialize `useForm` before you can start register your inputs, run this function before render.

React hook form use hook behind the scene by invoking `useForm`, you will receive the 4 methods.

Name | Type | Description

onSubmit | string | This is the default option, validation will trigger on submit, and then attach `onchange | blur | input` event listeners to re-validate them.

onBlur | string | Validation will trigger on each input `blur` event.

onChange | string | Not recommended as validation will go through each change on your input, and hence trigger render. Consider this as a bad performance practice.

## register: (args: Object):void

This is the function to register `Ref` and validation into `react-hook-form`. Validation rules are all based on html input/select's standard validation, however `react-hook-form` do allow custom validation with method `validate`.

Important: input name is required and unique for `react-hook-form` in order to register the input.

Name | Type | Description | Example

`ref` | `HTMLInputElement` | React element ref | ``` <input name="test" ref={ref => register({ ref }) } /> ```

`required` | `boolean` | A Boolean which, if true, indicates that the input must have a value before the form can be submitted. | ``` <input name="test" ref={ref => register({ ref, required: true }) } /> ```

`maxLength` | `number` | The maximum length of the value to accept for this input. | ``` <input name="test" ref={ref => register({ ref, maxLength: 2 }) } /> ```

`minLength` | `number` | The minimum length of the value to accept for this input. | ``` <input name="test" ref={ref => register({ ref, minLength: 1 }) } /> ```

`max` | `number` | The maximum value to accept for this input. | ``` <input name="test" ref={ref => register({ ref, max: 3 }) } /> ```

`min` | number | The minimum value to accept for this input. | ``` <input name="test" ref={ref => register({ ref, min: 3 }) } /> ```

`pattern` | `RegExp` | The regex pattern for the input. | ``` <input name="test" ref={ref => register({ ref, pattern: \[A-Za-z]{3}\ }) } /> ```

`validate` | `(Object) => Boolean | {[key: string]: (Object) => Boolean}` | You can pass a callback function with value as the argument to validate, or you can pass an object of callback functions to validate all of them. (refer to the examples) | ``` <input name="single" ref={ref => register({ ref, validate: (value) => value === '1' }) } /> <input name="multiple" ref={ref => register({ ref, validate: { isMoreThanOne: (value) => value > 1, isLessThanTen: (value) => value < 10, } }) } /> ```

## errors: {[key: string]: Object}

Object contain error info about the individual input.

## watch: (watchName: string | Array | undefined, defaultValue: string | Array | undefined) => string | number | boolean

Watch over input or selection change. Inital `watch` will always return `undefined`, because watch will run before the render fucntion . however you can set default value as the second argument, this is normally useful when you have some inital value to overwrite when `watch` return `undefined` on inital run.

Type | Description | Example

string | Target on individual input | `const value = watch('inputName');`

Array | Watch multiple inputs over the form | `const values = watch(['inputName1', 'inputName2']);`

undefined | Watch every input fields in the form | `const values = watch();`

## handleSubmit: ({ [key: string]: string | number | boolean }) => void

This function will pass you the form data when from validation is successful.
