- href<Path extends string>(path: Path, ...args: [] | [AnyParams]): string

Returns a resolved URL path for the specified route.

```
const h = href("/:lang?/about", { lang: "en" })
// -> `/en/about`

<Link to={href("/products/:id", { id: "abc123" })} />
```
