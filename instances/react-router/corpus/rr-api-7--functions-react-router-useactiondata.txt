# Function useActionData
# джерело: https://api.reactrouter.com/v7/functions/react-router.useActionData.html
# отримано: 2026-09-25
# версія: 7.18.4

- useActionData<T = any>(): undefined | SerializeFrom<T>

Returns the `action` data from the most recent `POST` navigation form submission or `undefined` if there hasn't been one.

#### Type Parameters

- T = any

#### Returns undefined | SerializeFrom<T>

The data returned from the route's `action` function, or `undefined` if no `action` has been called

#### Example

```
import { Form, useActionData } from "react-router";

export async function action({ request }) {
  const body = await request.formData();
  const name = body.get("visitorsName");
  return { message: `Hello, ${name}` };
}

export default function Invoices() {
  const data = useActionData();
  return (
    <Form method="post">
      <input type="text" name="visitorsName" />
      {data ? data.message : "Waiting..."}
    </Form>
  );
}

@public
```
