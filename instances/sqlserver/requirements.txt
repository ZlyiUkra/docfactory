# Пошук за змістом (необовʼязковий). ONNX замість torch: імпортується за секунди,
# без тисяч дрібних файлів. Без нього працює пошук лише по словах.
#
# Тут — збірка для відеокарти: ~144 тисячі фрагментів sql-docs і сотні тисяч архіву на процесорі
# рахувалися б понад добу (~5 фрагментів за секунду), на RTX 2070 у WSL — ~500 за секунду.
# fastembed-gpu — той самий модуль fastembed, але з onnxruntime-gpu замість onnxruntime (обидва
# разом не ставити: вони пишуть ту саму теку); onnxruntime-gpu ~340 МБ, а extras cuda і cudnn тягнуть
# колеса nvidia-* (CUDA 13 runtime, cuBLAS, cuFFT, cuRAND, nvJitLink, nvrtc ~1,2 ГБ; cuDNN 9 ~0,9 ГБ).
# Драйвер NVIDIA потрібен у Windows (у WSL — /usr/lib/wsl/lib). Без відеокарти onnxruntime-gpu
# рахує на процесорі, тими самими числами, що й звичайний onnxruntime (common/embed.py обирає сам).

onnxruntime-gpu[cuda,cudnn]==1.31.0
