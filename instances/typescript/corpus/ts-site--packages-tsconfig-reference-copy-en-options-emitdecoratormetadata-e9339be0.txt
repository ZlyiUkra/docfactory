# tsconfig-reference/copy/en/options/emitDecoratorMetadata
# джерело: https://github.com/microsoft/TypeScript-Website/blob/78d3b56c69ef1ee46453d291c09db4985b7e931e/packages/tsconfig-reference/copy/en/options/emitDecoratorMetadata.md
# отримано: 2026-09-29
# версія: 4.0.2

Enables experimental support for emitting type metadata for decorators which works with the module `reflect-metadata`.

For example, here is the JavaScript

```ts
// @experimentalDecorators
function LogMethod(target: any, propertyKey: string | symbol, descriptor: PropertyDescriptor) {
  console.log(target);
  console.log(propertyKey);
  console.log(descriptor);
}

class Demo {
  @LogMethod
  public foo(bar: number) {
    // do nothing
  }
}

const demo = new Demo();
```

With `emitDecoratorMetadata` not set to true (default):

```ts
// @experimentalDecorators
// @showEmit
function LogMethod(target: any, propertyKey: string | symbol, descriptor: PropertyDescriptor) {
  console.log(target);
  console.log(propertyKey);
  console.log(descriptor);
}

class Demo {
  @LogMethod
  public foo(bar: number) {
    // do nothing
  }
}

const demo = new Demo();
```

With `emitDecoratorMetadata` set to true:

```ts
// @experimentalDecorators
// @showEmit
// @emitDecoratorMetadata
function LogMethod(target: any, propertyKey: string | symbol, descriptor: PropertyDescriptor) {
  console.log(target);
  console.log(propertyKey);
  console.log(descriptor);
}

class Demo {
  @LogMethod
  public foo(bar: number) {
    // do nothing
  }
}

const demo = new Demo();
```
