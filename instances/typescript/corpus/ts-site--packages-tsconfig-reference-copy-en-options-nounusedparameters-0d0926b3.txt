# джерело: https://github.com/microsoft/TypeScript-Website/blob/78d3b56c69ef1ee46453d291c09db4985b7e931e/packages/tsconfig-reference/copy/en/options/noUnusedParameters.md
# отримано: 2026-09-29
# версія: 4.0.2

Report errors on unused parameters in functions.

```ts
// @noUnusedParameters
// @errors: 6133
const createDefaultKeyboard = (modelID: number) => {
  const defaultModelID = 23;
  return { type: "keyboard", modelID: defaultModelID };
};
```
