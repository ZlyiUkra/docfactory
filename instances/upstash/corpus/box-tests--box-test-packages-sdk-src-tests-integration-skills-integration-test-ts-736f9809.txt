## skills

```ts
describe.skipIf(!UPSTASH_BOX_API_KEY)("skills", () => {
  let box: Box;

  beforeAll(async () => {
    box = await Box.create({
      apiKey: UPSTASH_BOX_API_KEY!,
      agent: { provider: Agent.ClaudeCode, model: ClaudeCode.Opus_4_6 },
      skills: ["upstash/qstash-js/qstash-js"],
    });
  }, 120000);

  afterAll(async () => {
    try {
      await box?.delete();
    } catch {
      // cleanup best-effort
    }
  }, 30000);
```

### agent can use an installed skill

```ts
  it.skip("agent can use an installed skill", async () => {
```
