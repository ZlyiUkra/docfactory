export interface WorkflowLog {
  workflowRunId: string;
  workflowUrl: string;
  workflowState: "RUN_STARTED" | "RUN_SUCCESS" | "RUN_FAILED" | "RUN_CANCELED";
  workflowRunCreatedAt: number;
  workflowRunCompletedAt?: number;
  steps: Array<{
    steps: Array<{
      stepId?: number;
      stepName: string;
      stepType: string;
      callType: string;
      messageId: string;
      out?: string;
      concurrent: number;
      state: string;
      createdAt: number;
    }>;
    type: "sequential" | "parallel";
  }>;
}
