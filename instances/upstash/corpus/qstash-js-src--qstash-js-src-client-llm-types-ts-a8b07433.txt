## type ChatRequestCommonFields

```ts
type ChatRequestCommonFields = {
  frequency_penalty?: number;
  logit_bias?: Record<string, number>;
  logprobs?: boolean;
  top_logprobs?: number;
  max_tokens?: number;
  n?: number;
  presence_penalty?: number;
  response_format?: ChatResponseFormat;
```

```ts
type PromptChatRequestFields = ChatRequestCommonFields & {
```
