```ts
export type Endpoint = {
  /**

  /**

};
```

```ts
export type AddEndpointsRequest = {

   * Must be unique and only contain alphanumeric, hyphen, underscore and periods.
   */

  endpoints: Endpoint[];
};
```

```ts
  endpoints: (
    | {

        name?: string;
        url: string;
```

## type Topic

```ts
export type Topic = {
  /**
   * The name of this topic.
   */
  name: string;

  /**
   * A list of all subscribed endpoints
   */
  endpoints: Endpoint[];
};
```

## class Topics

```ts
export class Topics {
  private readonly http: Requester;

  constructor(http: Requester) {
    this.http = http;
  }
```

### Topics.addEndpoints()

```ts
  /**
   * Create a new topic with the given name and endpoints
   */
  public async addEndpoints(req: AddEndpointsRequest): Promise<void> {
    await this.http.request<Topic>({
      method: "POST",
      path: ["v2", "topics", req.name, "endpoints"],
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ endpoints: req.endpoints }),
      parseResponseAsJson: false,
    });
  }
```

### Topics.removeEndpoints()

```ts
  /**
   * Remove endpoints from a topic.
   */
  public async removeEndpoints(req: RemoveEndpointsRequest): Promise<void> {
    await this.http.request<Topic>({
      method: "DELETE",
      path: ["v2", "topics", req.name, "endpoints"],
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ endpoints: req.endpoints }),
      parseResponseAsJson: false,
    });
  }

  /**
   * Get a list of all topics.
   */
  public async list(): Promise<Topic[]> {
    return await this.http.request<Topic[]>({
      method: "GET",
      path: ["v2", "topics"],
    });
  }

  /**
   * Get a single topic
   */
  public async get(name: string): Promise<Topic> {
    return await this.http.request<Topic>({
      method: "GET",
      path: ["v2", "topics", name],
    });
  }
```

### Topics.delete()

```ts
  /**
   * Delete a topic
   */
  public async delete(name: string): Promise<void> {
    return await this.http.request<void>({
      method: "DELETE",
      path: ["v2", "topics", name],
      parseResponseAsJson: false,
    });
  }
}
```
