## type HistoryConfig

};

## class HistoryService

```ts
export class HistoryService {
  service: UpstashRedisHistory | InMemoryHistory;

  constructor(fields?: HistoryConfig) {
    this.service = fields?.redis
      ? new UpstashRedisHistory({
          client: fields.redis,
        })
      : new InMemoryHistory();
  }
}
```
