## the server is running

```ts
test(
  "the server is running",
  async () => {
    const res = await fetch(deploymentURL);

    expect(res.status).toEqual(200);

    const text = await res.text();

    console.log("Server returned >", text, "<");
  },
  {
    timeout: 20_000,
  }
);
```
