## const Spinner

```tsx
export const Spinner = ({
  isLoading,
  className,
  isLoadingText,
  children,
}: PropsWithChildren<SpinnerProps>) => {
  return (
    <div className={className ?? "flex items-center"}>
      {isLoading ? (
        <>
          {isLoadingText}
          <svg
            xmlns="http://www.w3.org/2000/svg"
            width="24"
            height="24"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
            strokeLinecap="round"
            strokeLinejoin="round"
            className={cn("h-4 w-4 animate-spin", isLoadingText ? "ml-2" : "")}
          >
            <path d="M21 12a9 9 0 1 1-6.219-8.56" />
          </svg>
        </>
      ) : (
        children
      )}
    </div>
  )
}
```
