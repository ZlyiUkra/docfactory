```ts
import { AnalyticsBackendClient } from "@upstash/redis-analytics";
import { CONFIG, type CustomEvents } from "./config";

let backendClient: AnalyticsBackendClient<CustomEvents> | null = null;
```

## function getAnalyticsBackendClient

```ts
export function getAnalyticsBackendClient(): AnalyticsBackendClient<CustomEvents> {
  if (!backendClient) {
    backendClient = new AnalyticsBackendClient<CustomEvents>({
      redis: {
        url: process.env.UPSTASH_REDIS_REST_URL!,
        token: process.env.UPSTASH_REDIS_REST_TOKEN!,
      },
      config: CONFIG,
    });
  }
  return backendClient;
}
```
