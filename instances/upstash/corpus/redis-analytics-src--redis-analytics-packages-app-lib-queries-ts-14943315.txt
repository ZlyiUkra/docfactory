```ts
import { useQuery } from "@tanstack/react-query";
import type { AnalyticsClient } from "@upstash/redis-analytics";
import type { CustomEvents, FeatureFlags } from "./config";
```

## function useSession

```ts
export function useSession(
  analytics: AnalyticsClient<CustomEvents, FeatureFlags>,
  sessionId: string | null
) {
  return useQuery({
    queryKey: ["session", sessionId],
    queryFn: () => analytics.getSession(sessionId!),
    enabled: !!sessionId,
  });
}
```

## function useFeatureFlag

```ts
export function useFeatureFlag<K extends string & keyof FeatureFlags>(
  analytics: AnalyticsClient<CustomEvents, FeatureFlags>,
  sessionId: string | null,
  flagName: K
) {
  return useQuery({
    queryKey: ["featureFlag", sessionId, flagName],
    queryFn: () => analytics.getFeatureFlag(sessionId!, flagName),
    enabled: !!sessionId,
  });
}
```
