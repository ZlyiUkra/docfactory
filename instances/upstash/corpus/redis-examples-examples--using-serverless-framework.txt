## Deploy with Serverless Framework

handler.js > ioredis async

handlerSync.js > ioredis sync

handlerEphemeral.js > does not reuse the connection

handlerNodeRedis.js > noderedis sync

## handler.js

```js
'use strict';

var Redis = require("ioredis");

let time = new Date();

if (typeof client === 'undefined') {
    var client = new Redis(process.env.REDIS_URL);
}

module.exports.hello = async(event) => {
    console.log("hello")
    await client.set("hello", "world");
    let response = await client.get("hello");
    return {response : response + "-" + time}
};
```

## handlerEphemeral.js

```js
'use strict';

var Redis = require("ioredis");

module.exports.hello = async(event) => {
    var client = new Redis(process.env.REDIS_URL);
    await client.set("hello", "world");
    let response = await client.get("hello");
    await client.quit();
    return {response : response}
};
```

## handlerNodeRedis.js

```js
'use strict';

var Redis = require("redis");


if (typeof client === 'undefined') {
    var client = Redis.createClient ({
        host : 'us1-full-bug-31874.upstash.io',
        port : '31874',
        password: '12345'
    });
}

module.exports.hello = (event, context, callback) => {
    context.callbackWaitsForEmptyEventLoop = false;

    // client.on("error", function(err) {

    // });
    client.set('foo','bar', function(err, dat) {
        console.log(dat)

```

## handlerSync.js

```js
'use strict';

var Redis = require("ioredis");


if (typeof client === 'undefined') {
    var client = new Redis("redis://:6f9ae50f60db42d587ada7174f2f64d0@eu1-apt-bedbug-32176.upstashdev.com:32176");
}

module.exports.hello = (event, context, callback) => {
    // This will allow us to freeze open connections to a database
    context.callbackWaitsForEmptyEventLoop = false;

    client.get("foo").then( (dat, err) => {
        callback(null, { response : dat })
    } )
};
```
