  constructor(key: string, start?: number, end?: number) {
    const command: unknown[] = ["bitcount", key]
    if (typeof start === "number") {
      command.push(start)
    }
    if (typeof end === "number") {
      command.push(end)
    }
    super(command)
  }
}
