  constructor(key: string, cursor: number, opts?: ScanCommandOptions) {
    const command = ["hscan", key, cursor];
    if (opts?.match) {
      command.push("match", opts.match);
    }
    if (typeof opts?.count === "number") {
      command.push("count", opts.count);
    }

    super(command);
  }
}
