  constructor(
    cmd:
      | [
          key: string,
          opts?: { indent?: string; newline?: string; space?: string },
          ...path: string[],

    const command = ["JSON.GET"];

        if (cmd[1].newline) {
          command.push("NEWLINE", cmd[1].newline);
        }
        if (cmd[1].space) {
          command.push("SPACE", cmd[1].space);
        }
      }
      // @ts-ignore - we know this is a string
      command.push(...cmd.slice(2));
    }

    super(command, opts);
  }
