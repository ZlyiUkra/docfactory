    cmd: [key: string, element: unknown, opts?: { rank?: number; count?: number; maxLen?: number }],
    opts?: CommandOptions<TData, TData>
  ) {

    }

      args.push("maxLen", cmd[2].maxLen);
    }
    super(args, opts);
