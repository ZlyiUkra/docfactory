### ScanCommand.constructor()

```ts
  constructor(cursor: number, opts?: ScanCommandOptions) {
    const command = ["scan", cursor]
    if (opts?.match) {
      command.push("match", opts.match)
    }
    if (typeof opts?.count === "number") {
      command.push("count", opts.count)
    }
    super(command)
  }
}
```
