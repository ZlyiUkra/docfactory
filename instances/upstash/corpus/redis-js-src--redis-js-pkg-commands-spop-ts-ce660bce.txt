### SPopCommand.constructor()

```ts
  constructor(
    [key, count]: [key: string, count?: number],
    opts?: CommandOptions<string | null, TData | null>,
  ) {
    const command: unknown[] = ["spop", key];
    if (typeof count === "number") {
      command.push(count);
    }
    super(command, opts);
  }
}
```
