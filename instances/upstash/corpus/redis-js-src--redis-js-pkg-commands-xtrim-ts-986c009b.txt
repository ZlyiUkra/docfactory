  constructor(
    [key, options]: [key: string, options: XTrimOptions],
    opts?: CommandOptions<number, number>
  ) {
    const { limit, strategy, threshold, exactness = "~" } = options;

    super(
      [
        "XTRIM",
        key,
        strategy,
        exactness,
        threshold,
        ...(limit ? ["LIMIT", limit] : []),
      ],
      opts
