### ZRangeCommand.constructor()

```ts
  constructor(
    [key, min, max, opts]: [
      key: string,
      min: number | string,
      max: number | string,
      opts?: ZRangeCommandOptions,
    ],
    cmdOpts?: CommandOptions<string[], TData>
  ) {
    const command: unknown[] = ["zrange", key, min, max];

    // Either byScore or byLex is allowed
    if (opts?.byScore) {
      command.push("byscore");
    }
    if (opts?.byLex) {
      command.push("bylex");
    }
    if (opts?.rev) {
      command.push("rev");
    }
    if (opts?.count !== undefined && opts.offset !== undefined) {
      command.push("limit", opts.offset, opts.count);
    }
    if (opts?.withScores) {
      command.push("withscores");
    }
    super(command, cmdOpts);
  }
}
```
