## type UpstashRequest

```ts
export type UpstashRequest = {
  path?: string[];
  /**
   * Request body will be serialized to json
   */
  body?: unknown;
};
export type UpstashResponse<TResult> = { result?: TResult; error?: string };
```

## interface Requester

```ts
export interface Requester {
  request: <TResult = unknown>(
    req: UpstashRequest,
  ) => Promise<UpstashResponse<TResult>>;
}
```

## type HttpClientConfig

```ts
export type HttpClientConfig = {
  headers?: Record<string, string>;
  baseUrl: string;
  options?: { backend?: string };
};
```

## class HttpClient

```ts
export class HttpClient implements Requester {
  public baseUrl: string;
  public headers: Record<string, string>;
  public readonly options?: { backend?: string };
```

### HttpClient.constructor()

```ts
  public constructor(config: HttpClientConfig) {
    this.baseUrl = config.baseUrl.replace(/\/$/, "");

    this.headers = { "Content-Type": "application/json", ...config.headers };

    this.options = config.options;
  }
```

### HttpClient.request()

```ts
  public async request<TResult>(
    req: UpstashRequest,
  ): Promise<UpstashResponse<TResult>> {
    if (!req.path) {
      req.path = [];
    }

    const requestOptions: RequestInit & { backend?: string } = {
      method: "POST",
      headers: this.headers,
      body: JSON.stringify(req.body),
      keepalive: true,

      /**
       * Fastly specific
       */
      backend: this.options?.backend,
    };

    // fetch is defined by isomorphic fetch
    // eslint-disable-next-line no-undef
    const res = await fetch(
      [this.baseUrl, ...req.path].join("/"),
      requestOptions,
    );
    const body = (await res.json()) as UpstashResponse<TResult>;
    if (!res.ok) {
      throw new UpstashError(body.error!);
    }

    return body;
  }
}
```
