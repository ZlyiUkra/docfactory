test("works", async () => {
  console.log({ deploymentURL });
  const res = await fetch(deploymentURL);
  const body = await res.text();
  console.log({ body });
  expect(res.status).toBe(200);
  const json = JSON.parse(body) as { counter: number };
  expect(typeof json.counter).toBe("number");
});
