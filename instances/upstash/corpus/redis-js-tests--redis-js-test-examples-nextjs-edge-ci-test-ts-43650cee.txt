```ts
test("works", async () => {
  const url = `${deploymentURL}/api/counter`;
  console.log({ url });

  const res = await fetch(url);
  expect(res.status).toEqual(200);
  const { counter } = (await res.json()) as { counter: number };
  expect(typeof counter).toEqual("number");
});
```
