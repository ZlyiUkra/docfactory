test("works", async () => {
  console.log({ deploymentURL });
  const url = `${deploymentURL}/api/hello`;
  const res = await fetch(url);
  expect(res.status).toEqual(200);
  const json = (await res.json()) as { count: number };
  expect(typeof json.count).toEqual("number");
});
