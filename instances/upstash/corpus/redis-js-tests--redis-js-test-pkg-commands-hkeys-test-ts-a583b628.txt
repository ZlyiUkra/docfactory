### returns all keys

```ts
  it("returns all keys", async () => {
    const key = newKey()
    const kv = {
      [randomUUID()]: randomUUID(),
      [randomUUID()]: randomUUID(),
    }
    await new HMSetCommand(key, kv).exec(client)
    const res = await new HKeysCommand(key).exec(client)
    expect(res.sort()).toEqual(Object.keys(kv).sort())
  })
})
```
