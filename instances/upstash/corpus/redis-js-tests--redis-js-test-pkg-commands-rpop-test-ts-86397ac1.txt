### returns the first element

```ts
  it("returns the first element", async () => {
    const key = newKey()
    const value = randomUUID()
    await new LPushCommand(key, value).exec(client)
    const res = await new RPopCommand(key).exec(client)
    expect(res).toEqual(value)
  })
})
```
