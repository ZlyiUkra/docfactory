### and

  it('and', async () => {
    const key1 = nanoid();
    const key2 = nanoid();
    const dest = nanoid();

    await set(key1, 'ali');
    await set(key2, 'veli');

    const { data } = await bitop('AND', dest, [key1, key2]);
    expect(data).toBe(4);
  });
