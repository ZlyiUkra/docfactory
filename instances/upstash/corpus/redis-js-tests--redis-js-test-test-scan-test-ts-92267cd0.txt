### basic

```ts
  it('basic', async () => {
    await flushdb();

    await mset(['key1', '1', 'key2', '2']);

    const { data } = await scan(0);
    expect(data[0]).toBe('0');
    expect(data[1].length).toBeGreaterThanOrEqual(2);
```

### with match

```ts
  it('with match', async () => {
    await flushdb();

    await mset(['key1', '1', 'key2', '2', 'mykey1', '3']);

    const { data } = await scan(0, { match: 'key*' });

    expect(data[1].length).toBeGreaterThanOrEqual(2);
  });
```
