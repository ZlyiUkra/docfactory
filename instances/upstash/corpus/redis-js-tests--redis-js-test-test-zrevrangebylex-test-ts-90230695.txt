### basic

```ts
  it('basic', async () => {
    const key = nanoid();

    const { data: addData } = await zadd(key, [
      '0',
      'a',
      '0',
      'b',
      '0',
      'c',
      '0',
      'd',
      '0',
      'e',
    ]);
    expect(addData).toBe(5);

    const { data: range1 } = await zrevrangebylex(key, '[c', '-');
    expect(range1).toMatchObject(['c', 'b', 'a']);
  });
```

### with limit

```ts
  it('with limit', async () => {
    const key = nanoid();

    const { data: addData } = await zadd(key, [
      '0',
      'a',
      '0',

    ]);
    expect(addData).toBe(5);

    const { data: range1 } = await zrevrangebylex(key, '[c', '-', 1, 2);

  });
});
```
