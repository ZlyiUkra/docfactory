import { addMessageToHistory, getMessageHistory, redis } from "@/utils"
import { createOpenRouter } from "@openrouter/ai-sdk-provider"
import { convertToModelMessages, streamText, UIMessage } from "ai"
import { serve } from "@upstash/workflow/nextjs"

## const openrouter

```ts
const openrouter = createOpenRouter({
  apiKey: process.env.OPENROUTER_API_KEY,
})

export const { POST } = serve(
  async (workflow) => {
    const { message, chatId, responseId, unix } = workflow.requestPayload as {
      message: UIMessage
      chatId: string
      responseId: string
      unix: number
    }

    // idempotent (!) function to add message to history instantly
    await addMessageToHistory(chatId, unix, message)

    const streamKey = `stream:${responseId}`

    await workflow.run("generate-llm-response", async () => {
      const history = await getMessageHistory(chatId)

      const messages = [...history, message]

      const stream = streamText({
        model: openrouter.chat("google/gemini-2.5-flash"),
        messages: convertToModelMessages(messages),
      })

      const uiMessageStream = stream.toUIMessageStream({
        generateMessageId: () => responseId,
        onFinish: async (result) => {
          const historyPipeline = redis.pipeline()

          result.messages.forEach((message) => {
            historyPipeline.zadd(`chat:history:${chatId}`, {
              score: Date.now(),
              member: message,
            })
          })

          await historyPipeline.exec()
        },
      })

      for await (const message of uiMessageStream) {
        const pipeline = redis.pipeline()
        pipeline.xadd(streamKey, "*", message, {})
        pipeline.publish(streamKey, null)
        await pipeline.exec()
      }
    })
  },
  { retries: 0 }
)
```
