```ts
export async function POST(req: NextRequest) {
  const { responseId, isInHistory } = (await req.json()) as {
    responseId: string
    chatId: string
    isInHistory: boolean
  }

  const streamKey = `stream:${responseId}`
  const groupName = `sse-group-${nanoid()}`

  if (isInHistory) {
    return new Response(
      new ReadableStream({
        start(controller) {
          controller.close()
        },
      }),
      { headers }
    )
  }

  const subscription = redis.subscribe(streamKey)

  await redis
    .xgroup(streamKey, {
      options: { MKSTREAM: true },
      type: "CREATE",
      group: groupName,
      id: "0",
    })
    .catch(() => {
      // group already exists, no problem
    })

  const response = new Response(
    new ReadableStream({
      async start(controller) {
        const readStreamMessages = async () => {
          const chunks = await redis.xreadgroup(groupName, `consumer-1`, streamKey, ">")

          if (chunks && chunks.length > 0) {
            const [_, messages] = chunks[0] as [string, any]
            for (const [_, fields] of messages) {
              const chunk = toObject(fields) as Chunk
              controller.enqueue(json(chunk))
            }
          }
        }

        // read initial messages
        await readStreamMessages()

        // every time we `pipeline.publish` in producer, we consume here
        subscription.on("message", async () => {
          await readStreamMessages().catch((err) => {
            console.log("error reading message:", err)
          })
        })

        req.signal.addEventListener("abort", () => {
          subscription.unsubscribe()
          controller.close()
        })
      },
    }),
    { headers }
  )

  return response
}
```
