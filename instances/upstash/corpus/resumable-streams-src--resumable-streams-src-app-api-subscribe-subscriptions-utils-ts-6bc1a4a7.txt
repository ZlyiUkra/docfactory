## function toObject

```ts
export function toObject(arr: any) {
  const obj: Record<string, any> = {}

  for (let i = 0; i < arr.length; i += 2) {
    obj[arr[i]] = arr[i + 1]
  }

  return obj as Chunk
}
```

## const json

```ts
export const json = (data: Record<string, unknown>) => {
  return new TextEncoder().encode(`data: ${JSON.stringify(data)}\n\n`)
}
```
