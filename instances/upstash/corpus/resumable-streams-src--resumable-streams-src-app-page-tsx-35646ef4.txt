```tsx
import { Chat } from "@/components/chat"
import { getMessageHistory } from "@/utils"
import { UIMessage } from "ai"
import z from "zod"
```

## const searchParamSchema

```tsx
const searchParamSchema = z
  .object({
    chatId: z.string().optional(),
    responseId: z.string().optional(),
  })
  .loose()
```

## type Params

```tsx
type Params = Promise<{ chatId?: string; responseId?: string }>
```

## const Page

```tsx
const Page = async ({ searchParams }: { searchParams: Params }) => {
  const params = await searchParams
  const { chatId, responseId } = searchParamSchema.parse(params)

  let messages: Record<string, UIMessage[]> = {}
  let isInHistory: boolean = false

  if (chatId) {
    messages[chatId] = await getMessageHistory(chatId)
    isInHistory = messages[chatId].some((message) => message.id === responseId)
  }

  return <Chat initialMessages={messages ?? []} isInHistory={isInHistory} />
}

export default Page
```
