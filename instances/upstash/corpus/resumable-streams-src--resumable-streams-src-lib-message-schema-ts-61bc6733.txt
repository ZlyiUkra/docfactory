```ts
import { z } from "zod"
```

## const Status

```ts
export const Status = z
  .enum(["idle", "loading", "streaming", "completed", "error"])
  .default("idle")
```

## type GenerationStatus

```ts
export type GenerationStatus = z.infer<typeof Status>
```

## const Chunk

```ts
export const Chunk = z.object({
  chunk: z.string(),
})
```

## type ChunkMessage

```ts
export type ChunkMessage = z.infer<typeof Chunk>
```

## const Metadata

```ts
export const Metadata = z.object({ type: z.literal("metadata"), status: Status })
```

## type MetadataMessage

```ts
export type MetadataMessage = z.infer<typeof Metadata>
```
