## const addMessageToHistory

```ts
export const addMessageToHistory = async (
  chatId: string,
  unix: number,
  message: UIMessage
): Promise<void> => {
  const messageKey = `${unix}-${message.id || Math.random()}`

  await redis.zadd(
    `chat:history:${chatId}`,
    // 👇 this is important for being idempotent (if exists, do nothing)
    { nx: true },
    { score: unix, member: message }
  )
}
```

## const getMessageHistory

```ts
export const getMessageHistory = async (chatId: string): Promise<UIMessage[]> => {
  const messages = await redis.zrange<UIMessage[]>(`chat:history:${chatId}`, 0, -1)

  if (!messages || messages.length === 0) {
    return []
  }

  return messages
}
```
