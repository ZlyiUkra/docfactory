## const getStrategyKey

```ts
export const getStrategyKey = (strategy: Strategy) => {
	return `${strategy.methods.join("/")}-${strategy.path}`
}
```

```ts
export const parseRatelimitIdentifier = (identifierKey: string) => {
	const res = identifierKey.split(".")
```
