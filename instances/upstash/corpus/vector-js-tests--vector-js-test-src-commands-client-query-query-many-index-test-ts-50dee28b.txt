```ts
  test("should query in batches successfully", async () => {
    const ID = randomID();
    const initialData = [
      {
        id: `1-${ID}`,
        vector: range(0, 384),
        metadata: {
          animal: "elephant",
          tags: ["mammal"],
          diet: "herbivore",
        },
      },
      {
        id: `2-${ID}`,
        vector: range(0, 384),
        metadata: {
          animal: "tiger",
          tags: ["mammal"],
          diet: "carnivore",
        },
      },
    ];
    await new UpsertCommand(initialData).exec(client);

    await awaitUntilIndexed(client);

    const res = await new QueryManyCommand<{
      animal: string;
      tags: string[];
      diet: string;
    }>([
      {
        vector: initialData[0].vector,
        topK: 1,
        filter: "tags[0] = 'mammal' AND diet = 'herbivore'",
        includeMetadata: true,
      },
      {
        vector: initialData[1].vector,
        topK: 1,
        filter: "tags[0] = 'mammal' AND diet = 'carnivore'",
        includeMetadata: true,
      },
    ]).exec(client);

    expect(res).toEqual([
      [
        {
          id: `1-${ID}`,
          score: 1,
          metadata: { animal: "elephant", tags: ["mammal"], diet: "herbivore" },
        },
      ],
      [
        {
          id: `2-${ID}`,
          score: 1,
          metadata: { animal: "tiger", tags: ["mammal"], diet: "carnivore" },
        },
      ],
    ]);
  });
```

### should return nested array when single query is provided

```ts
  test("should return nested array when single query is provided", async () => {
    const ID = randomID();
    const initialData = [
      {
        id: `1-${ID}`,
        vector: range(0, 384),
        metadata: {
          animal: "elephant",
          tags: ["mammal"],
          diet: "herbivore",
        },
      },
    ];
    await new UpsertCommand(initialData).exec(client);

    await awaitUntilIndexed(client);

    const res = await new QueryManyCommand<{
      animal: string;
      tags: string[];
      diet: string;
    }>([
      {
        vector: initialData[0].vector,
        topK: 1,
        filter: "tags[0] = 'mammal' AND diet = 'herbivore'",
        includeMetadata: true,
      },
    ]).exec(client);

    // Should always return a nested array, even with a single query
    expect(Array.isArray(res)).toBe(true);
    expect(res.length).toBe(1);
    expect(Array.isArray(res[0])).toBe(true);
    expect(res[0][0]).toMatchObject({
      id: `1-${ID}`,
      score: 1,
      metadata: { animal: "elephant", tags: ["mammal"], diet: "herbivore" },
    });
  });
});
```

### should query hybrid index

```ts
  test("should query hybrid index", async () => {
    const namespace = "query-hybrid";
    await populateHybridIndex(hybridIndex, namespace);

    const result = await hybridIndex.queryMany(
      [
        {
          topK: 1,
          vector: [0.1, 0.1],
          sparseVector: {
            indices: [3, 4],
            values: [0.1, 0.2],
          },
          fusionAlgorithm: FusionAlgorithm.RRF,
        },
        {
          topK: 1,
          vector: [0.5, 0.1],
          sparseVector: {
            indices: [0, 1],
            values: [0.5, 0.1],
          },
          includeVectors: true,
        },
        {
          topK: 1,
          sparseVector: {
            indices: [2, 3],
            values: [0.5, 0.5],
          },
          weightingStrategy: WeightingStrategy.IDF,
          fusionAlgorithm: FusionAlgorithm.DBSF,
          includeMetadata: true,
        },
      ],
      {
        namespace,
      }
    );

    expect(result).toEqual([
      [
        {
          id: "id2",
          score: 0.033_333_335,
        },
      ],
      [
        {
          id: "id0",
          score: 0.032_795_697,
          vector: [0.1, 0.2],
          sparseVector: {
            indices: [0, 1],
            values: [0.1, 0.2],
          },
        },
      ],
      [
        {
          id: "id2",
          score: 0.266_666_38,
          metadata: {
            key: "value",
          },
        },
      ],
    ]);
  });
});
```
