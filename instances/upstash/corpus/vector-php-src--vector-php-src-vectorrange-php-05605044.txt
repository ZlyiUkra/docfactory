        public bool $includeVectors = false,
        public bool $includeData = false,

    public function toArray(): array

            'limit' => $this->limit,
            'cursor' => $this->cursor,
            'includeMetadata' => $this->includeMetadata,
            'includeVectors' => $this->includeVectors,
            'includeData' => $this->includeData,
        ];
    }
}
