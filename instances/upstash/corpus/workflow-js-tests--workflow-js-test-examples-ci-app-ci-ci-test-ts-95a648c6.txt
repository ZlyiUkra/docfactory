## workflow integration tests

```ts
describe("workflow integration tests", () => {
  TEST_ROUTES.forEach(testConfig => {
    test(
      testConfig.route,
      async () => {
        await initiateTest(testConfig.route, testConfig.waitForSeconds)
      },
      {
        timeout: (testConfig.waitForSeconds + 8) * 1000
      }
    )
  });
})
```
