# Спільне середовище фабрики, збірка для відеокарти (.venvs/gpu).
#
# Те саме об'єднання, що в requirements/cpu.txt, але замість fastembed — fastembed-gpu і onnxruntime-gpu
# з бібліотеками CUDA: common/embed.py сам бачить CUDA і рахує вектори на відеокарті (~500–800 фрагментів
# за секунду на RTX 2070 у WSL проти ~5 на процесорі). Збирається так, з кореня docfactory/:
#
#   python3 -m venv .venvs/gpu
#   .venvs/gpu/bin/python -m pip install -U pip
#   .venvs/gpu/bin/python -m pip install -r requirements/gpu.txt
#
# Python 3.10+. Версії й верхні межі ті самі, що в requirements.txt примірників зі збіркою для відеокарти.
# Без відеокарти onnxruntime-gpu рахує на процесорі тими самими числами, що й звичайний onnxruntime, тож
# цим середовищем можна користуватися й без неї (лише займає ~2,6 ГБ, ~2,8 ГБ з розпізнаванням).
#
# Другий крок збірки — розпізнавання сторінок PDF для читача cncfcurriculum (kubernetes), окремим файлом і
# обов'язково з --no-deps (чому — у шапці самого файла):
#
#   .venvs/gpu/bin/python -m pip install --no-deps -r requirements/gpu-ocr.txt

anthropic>=0.40,<1
python-dotenv>=1.0.0

# MCP-сервер і клієнт. У SDK 2.0 клас FastMCP перейменували на MCPServer —
# код підтримує обидві версії.
mcp[cli]>=2.1,<3

# fastembed-gpu — той самий модуль fastembed, але з onnxruntime-gpu замість onnxruntime (обидва
# разом не ставити: вони пишуть ту саму теку); onnxruntime-gpu ~340 МБ, а extras cuda і cudnn тягнуть
# колеса nvidia-* (CUDA 13 runtime, cuBLAS, cuFFT, cuRAND, nvJitLink, nvrtc ~1,2 ГБ; cuDNN 9 ~0,9 ГБ).
# Драйвер NVIDIA потрібен у Windows (у WSL — /usr/lib/wsl/lib).
fastembed-gpu==0.9.0
onnxruntime-gpu[cuda,cudnn]==1.31.0

# YAML у джерелах: читачі openapi, sdkspec, ghhistory, githubdocs, snooty (certbot, mongodb, supabase та інші).
PyYAML>=6,<7

# PDF із текстовим шаром (argon2, ecmascript, owasp, patterns, kubernetes).
pypdf>=6,<7

# Сторінки PDF як картинки для розпізнавання (kubernetes). Самого розпізнавача rapidocr-onnxruntime
# тут немає навмисно: він вимагає звичайний onnxruntime, а той у одному середовищі з onnxruntime-gpu
# затирає його файли. Його ставить другий крок збірки, requirements/gpu-ocr.txt, з --no-deps (див. вище).
pypdfium2>=4,<6
